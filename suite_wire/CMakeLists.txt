set( sources "00_wire_slots.cpp" "01_wire_bridges.cpp" )
foreach( source IN LISTS sources )
    get_filename_component( app_name "${source}" NAME_WE )
    motor_vs_src_dir( "${source}" )
    add_executable( ${app_name} "${source}" )
    target_link_libraries( ${app_name} PRIVATE motor::wire motor::memory motor::log )
    set_target_properties( ${app_name} PROPERTIES FOLDER "suites/suite_wire" )
endforeach()
target_link_libraries( 01_wire_bridges PRIVATE motor::graphics )

if( BUILD_TESTING )
    add_test( NAME wire_slots COMMAND 00_wire_slots )
    set_tests_properties( wire_slots PROPERTIES TIMEOUT 30 LABELS wire )
    foreach( scenario input output rebind )
        add_test( NAME wire_bridge_${scenario} COMMAND 01_wire_bridges ${scenario} )
        set_tests_properties( wire_bridge_${scenario} PROPERTIES
            TIMEOUT 30 LABELS wire SKIP_RETURN_CODE 77 )
    endforeach()
endif()
