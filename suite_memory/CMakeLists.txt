find_package( Threads REQUIRED )

foreach( source 00_memory_ownership.cpp 01_memory_allocations.cpp )
    get_filename_component( app_name "${source}" NAME_WE )
    motor_vs_src_dir( "${source}" )
    add_executable( ${app_name} "${source}" )
    target_link_libraries( ${app_name} PRIVATE motor::memory motor::log Threads::Threads )
    set_target_properties( ${app_name} PROPERTIES FOLDER "suites/suite_memory" )
endforeach()

if( BUILD_TESTING )
    foreach( scenario sharing guard concurrent )
        add_test( NAME memory_ownership_${scenario} COMMAND 00_memory_ownership ${scenario} )
        set_tests_properties( memory_ownership_${scenario} PROPERTIES TIMEOUT 30 LABELS memory )
    endforeach()
    foreach( scenario raw typed_array containers malloc_copy malloc_move malloc_assignment )
        add_test( NAME memory_allocations_${scenario} COMMAND 01_memory_allocations ${scenario} )
        set_tests_properties( memory_allocations_${scenario} PROPERTIES TIMEOUT 30 LABELS memory )
    endforeach()
endif()
