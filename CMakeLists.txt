cmake_minimum_required( VERSION 3.20 )
project( motor_ai_suites LANGUAGES C CXX )

set( CMAKE_CXX_STANDARD 20 )
set( CMAKE_CXX_STANDARD_REQUIRED ON )
set( CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" )
set( CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" )

add_subdirectory( motor )
list( APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/motor/cmake/macros" )
include( visual_studio_src_dir )

add_subdirectory( suite_graphics )
include( CTest )
add_subdirectory( suite_concurrent )
add_subdirectory( suite_memory )
add_subdirectory( suite_wire )
add_subdirectory( suite_core )
add_subdirectory( suite_math )
option( MOTOR_AI_BUILD_RAYLIB "Build the native raylib benchmark" OFF )
if( MOTOR_AI_BUILD_RAYLIB )
    add_subdirectory( externals/raylib )
    add_subdirectory( suite_raylib )
endif()
option( MOTOR_AI_BUILD_BGFX "Build the native bgfx benchmark" OFF )
option( MOTOR_AI_BUILD_CINDER "Build the native Cinder benchmark" OFF )
option( MOTOR_AI_BUILD_OPENGL "Build the direct OpenGL benchmark" OFF )
if( MOTOR_AI_BUILD_BGFX OR MOTOR_AI_BUILD_CINDER OR MOTOR_AI_BUILD_OPENGL )
    add_subdirectory( suite_reference )
endif()
if( MOTOR_AI_BUILD_BGFX OR MOTOR_AI_BUILD_OPENGL )
    add_subdirectory( externals/reference_glfw )
endif()
if( MOTOR_AI_BUILD_BGFX )
    add_subdirectory( externals/bgfx )
    add_subdirectory( suite_bgfx )
endif()
if( MOTOR_AI_BUILD_CINDER )
    add_subdirectory( externals/cinder )
    add_subdirectory( suite_cinder )
endif()
if( MOTOR_AI_BUILD_OPENGL )
    add_subdirectory( suite_opengl )
endif()
set_property( DIRECTORY PROPERTY VS_STARTUP_PROJECT 00_triangle )
